# Файл roles/common/tasks/main.yml
- name: Set timezone
  timezone:
    name: "{{ timezone }}"

- name: Install systemd-timesyncd
  package:
    name: systemd-timesyncd
    state: present

- name: Configure NTP servers
  lineinfile:
    path: /etc/systemd/timesyncd.conf
    regexp: "^{{ item.key }}="
    line: "{{ item.key }}={{ item.value }}"
    backup: yes
  loop:
    - { key: "NTP", value: "{{ ntp_servers | join(' ') }}" }
    - { key: "FallbackNTP", value: "2.pool.ntp.org 3.pool.ntp.org" }
  notify: restart timesyncd

- name: Ensure netplan.io is installed
  package:
    name: netplan.io
    state: present

- name: Write netplan configuration
  template:
    src: netplan.yaml.j2
    dest: /etc/netplan/01-netcfg.yaml
    mode: '0644'
  notify: apply netplan

- name: Install UFW
  package:
    name: ufw
    state: present

- name: Allow SSH and HTTP ports
  ufw:
    rule: allow
    port: "{{ item }}"
    proto: tcp
  loop:
    - 22
    - 80

- name: Set default policy
  ufw:
    direction: incoming
    policy: deny

- name: Enable UFW
  ufw:
    state: enabled

# Файл roles/postgresql_lab/vars/main.yml
postgresql_packages:
  - postgresql-{{ postgresql_version }}
  - postgresql-contrib-{{ postgresql_version }}
postgresql_data_dir: "/var/lib/postgresql/{{ postgresql_version }}"
primary_data_dir: "{{ postgresql_data_dir }}/primary"
standby_data_dir: "{{ postgresql_data_dir }}/standby"



# Файл roles/postgresql_lab/tasks/main.yml
- name: Install PostgreSQL
  apt:
    name: "{{ postgresql_packages }}"
    state: present
    update_cache: yes

- name: Drop default 'main' cluster if exists
  command: pg_dropcluster {{ postgresql_version }} main --stop
  register: drop_main
  failed_when: false
  changed_when: drop_main.rc == 0

- name: Create primary cluster
  command: pg_createcluster {{ postgresql_version }} primary --port {{ postgresql_port_primary }}
  args:
    creates: "{{ primary_data_dir }}"

- name: Create standby cluster (empty, will be overwritten)
  command: pg_createcluster {{ postgresql_version }} standby --port {{ postgresql_port_standby }} --start-conf=no
  args:
    creates: "{{ standby_data_dir }}"

- name: Stop standby cluster for cloning
  command: pg_ctlcluster {{ postgresql_version }} standby stop
  failed_when: false

- name: Configure primary postgresql.conf
  template:
    src: postgresql.conf.primary.j2
    dest: "/etc/postgresql/{{ postgresql_version }}/primary/postgresql.conf"
    owner: postgres
    group: postgres
    mode: '0644'
  notify: restart primary

- name: Configure primary pg_hba.conf for replication
  template:
    src: pg_hba.conf.primary.j2
    dest: "/etc/postgresql/{{ postgresql_version }}/primary/pg_hba.conf"
    owner: postgres
    group: postgres
    mode: '0640'
  notify: restart primary

- name: Create replicator user on primary
  become_user: postgres
  postgresql_user:
    db: postgres
    name: replicator
    password: "{{ replicator_password }}"
    role_attr_flags: REPLICATION,LOGIN
    port: "{{ postgresql_port_primary }}"
  notify: restart primary

- name: Ensure primary is running
  command: pg_ctlcluster {{ postgresql_version }} primary start

- name: Clean standby data directory
  file:
    path: "{{ standby_data_dir }}"
    state: absent

- name: Recreate empty standby directory
  file:
    path: "{{ standby_data_dir }}"
    state: directory
    owner: postgres
    group: postgres
    mode: '0700'

- name: Clone primary to standby using pg_basebackup
  become_user: postgres
  command: >
    pg_basebackup -h 127.0.0.1 -p {{ postgresql_port_primary }}
    -U replicator -D {{ standby_data_dir }} -R -P -X stream
  environment:
    PGPASSWORD: "{{ replicator_password }}"

- name: Configure standby postgresql.conf (hot_standby)
  template:
    src: postgresql.conf.standby.j2
    dest: "/etc/postgresql/{{ postgresql_version }}/standby/postgresql.conf"
    owner: postgres
    group: postgres
    mode: '0644'
  notify: restart standby

- name: Start standby
  command: pg_ctlcluster {{ postgresql_version }} standby start

- name: Wait for replication to start
  pause:
    seconds: 5

- name: Create database and user on primary
  become_user: postgres
  postgresql_db:
    name: "{{ db_name }}"
    owner: "{{ db_user }}"
    port: "{{ postgresql_port_primary }}"
  notify: restart primary

- name: Create db user on primary
  become_user: postgres
  postgresql_user:
    db: "{{ db_name }}"
    name: "{{ db_user }}"
    password: "{{ db_user_password }}"
    port: "{{ postgresql_port_primary }}"
    priv: ALL
  notify: restart primary

- name: Create books table on primary
  become_user: postgres
  postgresql_query:
    db: "{{ db_name }}"
    port: "{{ postgresql_port_primary }}"
    login_user: "{{ db_user }}"
    login_password: "{{ db_user_password }}"
    query: |
      CREATE TABLE IF NOT EXISTS books (
        id SERIAL PRIMARY KEY,
        title TEXT NOT NULL,
        year INTEGER CHECK (year BETWEEN 1450 AND EXTRACT(YEAR FROM CURRENT_DATE))
      );
  register: create_table

- name: Insert sample books
  become_user: postgres
  postgresql_query:
    db: "{{ db_name }}"
    port: "{{ postgresql_port_primary }}"
    login_user: "{{ db_user }}"
    login_password: "{{ db_user_password }}"
    query: |
      INSERT INTO books (title, year) VALUES
        ('Война и мир', 1869),
        ('1984', 1949),
        ('Гарри Поттер и философский камень', 1997)
      ON CONFLICT DO NOTHING;

- name: Verify replication – query standby
  become_user: postgres
  postgresql_query:
    db: "{{ db_name }}"
    port: "{{ postgresql_port_standby }}"
    login_user: "{{ db_user }}"
    login_password: "{{ db_user_password }}"
    query: "SELECT count(*) FROM books;"
  register: standby_check
  failed_when: standby_check.query_result[0].count | int == 0
  tags: test

- name: Copy backup script
  copy:
    src: backup_hw_db.sh
    dest: "{{ backup_dir }}/backup_hw_db.sh"
    owner: "{{ ansible_user }}"
    group: "{{ ansible_user }}"
    mode: '0755'
  tags: backup

- name: Ensure backup directory exists
  file:
    path: "{{ backup_dir }}"
    state: directory
    mode: '0750'

- name: Schedule daily backup via cron
  cron:
    name: "PostgreSQL backup of {{ db_name }}"
    minute: 0
    hour: 2
    job: "{{ backup_dir }}/backup_hw_db.sh"
    user: "{{ ansible_user }}"
  tags: backup

- name: Display pg_stat_replication on primary
  become_user: postgres
  command: psql -p {{ postgresql_port_primary }} -Atc "SELECT application_name, state FROM pg_stat_replication;"
  register: replication_status
  tags: info
  changed_when: false

- debug:
    msg: "Replication status: {{ replication_status.stdout }}"
  tags: info

# Плейбук playbooks/promote.yml
- name: Promote standby to primary and test
  hosts: target-vm
  gather_facts: yes
  tasks:
    - name: Promote standby cluster
      command: pg_ctlcluster {{ postgresql_version }} standby promote
      register: promote_result
      changed_when: "'server promoted' in promote_result.stdout"

    - name: Wait for promotion
      pause:
        seconds: 3

    - name: Check pg_is_in_recovery on promoted node (should be false)
      become_user: postgres
      command: psql -p {{ postgresql_port_standby }} -Atc "SELECT pg_is_in_recovery();"
      register: recovery_status
      failed_when: recovery_status.stdout != 'f'

    - debug:
        msg: "Promoted standby is now primary (pg_is_in_recovery = {{ recovery_status.stdout }})"
