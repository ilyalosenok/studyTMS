1. Функция с массивом

#!/bin/bash

homework12_top5() {
    local log_file="$1"
    
    if [[ ! -f "$log_file" ]]; then
        echo "Ошибка: файл $log_file не найден"
        return 1
    fi
    
    # Читаем файл через readarray
    readarray -t lines < "$log_file"
    
    # Ассоциативный массив для подсчета
    declare -A ip_count
    
    for line in "${lines[@]}"; do
        ip="${line%% *}"
        ((ip_count["$ip"]++))
    done
    
    for ip in "${!ip_count[@]}"; do
        echo "${ip_count[$ip]} $ip"
    done | sort -rn | head -5 | awk '{print $2 " - " $1 " запросов"}'
}

time homework12_top5 "$1"
