#!/usr/bin/env bash
# Скрипт для домашнего задания в нем есть несколько ошибок.
# Задача - найти и исправить их с помощью set -x, shellcheck и отладки в редакторе (или трапа DEBUG).
# Скрипт по замыслу: читает список URL из файла, для каждого выводит
# "проверка: URL" b "статус: ОК" (имитация проверки).
INPUT_FILE="${1:-urls.txt}"
MAX_PARALLEL=2
declare -a PIDS
process_url() {
local url=$1
echo "проверка: $url"
# Не определена переменная RESULT
if [ -n "$RESULT" ]; then
echo "результат: $RESULT"
fi 
sleep 1
echo "статус: ОК"
}
if [ !-f $INPUT_FILE ]; then
echo "Файл не найден: $INPUT_FILE" >&2
exit 1
fi 
count=0
while IFS= read -r line; do
# Строка ниже удаляет комментарии
line="${line%%#*}"
# Неправильная команда для удаления пробелов. Нашел что можно использовать команды line="${line#"${line%%[![ ]]*}"}" и line="${line%"${line##*[![ ]]}"}"
# или с помощью sed line="$(echo "$line" | sed -e 's/^[[ ]]*//' -e 's/[[ ]]*$//')"
line="${line##*()}"
line="${line%%*()}"
if [ -z "$line" ]; then
continue
fi
process_url $line &
PIDS+=($!)
count=$((count+1))
if [ $count -ge $MAX_PARALLEL ]; then
wait ${PIDS[0]}
PIDS=("${PIDS[@]:1}")
count=$((count-1))
fi
done < $INPUT_FILE
wait ${PIDS[@]}
echo "Готово."
