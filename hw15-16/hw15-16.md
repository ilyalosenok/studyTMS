# 1. HTTP: nginx и статический сайт
# Каталог с index.html
sudo mkdir -p /var/www/hw-ilyalosenok/html
nano index.html
# Заполняем файл:
    <!DOCTYPE html>
    <html>
    <head>
        <title>Homework 15-16</title>
    </head>
    <body>
        <h1>Лосенок Илья Юрьевич</h1>
        <p>Дата: 26.05.2026 г.</p>
    </body>
    </html>
sudo chmod -R 755 /var/www/hw-ilyalosenok/html
# Создаем виртуальный хост: 
cd /etc/nginx/sites-available
nano hw-ilyalosenok.local
# Заполняем конфиг:
server {
    listen 80;
    server_name hw-ilyalosenok.local;
    root /var/www/hw-ilyalosenok/html;
    index index.html;

    location / {
        try_files $uri $uri/ =404;
    }
}
# Активируем хост
sudo ln -s /etc/nginx/sites-available/hw-ilyalosenok.local /etc/nginx/sites-enabled/

echo "192.168.0.50 hw-ilyalosenok.local" | sudo tee -a /etc/hosts
curl -v http://hw-ilyalosenok.local

# 2. OPENSSL: просмотр сертификата
echo | openssl s_client -connect google.com:443 -servername google.com 2>/dev/null | openssl x509 -noout -subject -issuer -dates
# вывод команды:
subject=CN = *.google.com
issuer=C = US, O = Google Trust Services, CN = WE2
notBefore=May  7 15:51:36 2026 GMT
notAfter=Jul 30 15:51:35 2026 GMT
# -servername используется для передачи Server Name Indication. Это помогает избежать ошибки проверки имени хоста, когда на одном ip-адресе и порту размещено несколько хостов. 

# 3. HTTPS на nginx
openssl req -x509 -newkey rsa:4096 -sha256 -days 365 -nodes \
  -keyout hw-ilyalosenok.local.key \
  -out hw-ilyalosenok.local.crt \
  -subj "/CN=hw-ilyalosenok.local" \
  -addext "subjectAltName=DNS:hw-ilyalosenok.local"

  sudo mkdir -p /etc/nginx/ssl
sudo cp hw-ilyalosenok.local.crt /etc/nginx/ssl/
sudo cp hw-ilyalosenok.local.key /etc/nginx/ssl/
sudo chmod 600 /etc/nginx/ssl/hw-ilyalosenok.local.key

# конфиг:                                                                                                     
server {
    listen 80;
    server_name hw-ilyalosenok.local;
    return 301 https://$server_name$request_uri;
}

server {
    listen 443 ssl http2;
    server_name hw-ilyalosenok.local;

    ssl_certificate     /etc/nginx/ssl/hw-ilyalosenok.local.crt;
    ssl_certificate_key /etc/nginx/ssl/hw-ilyalosenok.local.key;

    ssl_protocols TLSv1.2 TLSv1.3;

    root /var/www/html;
    index index.html index.htm;
}

# без -k браузер\curl ругаются на самоподписанный сертификат потому что он не подписан ни одним из удостоверяющих центров сертификации (CA).
