CREATE TABLE books (

# Собрать лабораторию primary + standby по материалам занятия. 
sudo pg_createcluster 16 primary --port 5432
sudo pg_createcluster 16 standby --port 5433
# Настройка primary
# Редактируем postgresql.conf
sudo nano /etc/postgresql/16/primary/postgresql.conf
# Раскомментируем строки:
listen_addresses = '*'
wal_level = replica
max_wal_senders = 10
max_replication_slots = 10
wal_keep_size = 2000
# Настраиваем pg_hba.conf для репликации
sudo nano /etc/postgresql/16/primary/pg_hba.conf
# Добавляем в конец конфига строку:
host    replication     replicator      127.0.0.1/32            md5

sudo pg_ctlcluster 16 primary restart
sudo -u postgres psql -p 5432 -c "CREATE ROLE replicator WITH REPLICATION LOGIN PASSWORD 'repl';"

# Настройка standby реплики
sudo pg_ctlcluster 16 standby stop
sudo rm -rf /var/lib/postgresql/16/standby/*
sudo -u postgres pg_basebackup -h 127.0.0.1 -p 5432 -U replicator -D /var/lib/postgresql/16/standby/ -R -P
sudo pg_ctlcluster 16 standby start

# После промоута новый бывшая replica начинает работать на своём таймлайне и может принять записи, которых нет на старом primary. Если включить его без синхронизации, таймлайны разойдутся, и кластер потеряет согласованность — возможна потеря данных или ошибки репликации.

# Настроить логический бэкап своей базы, сделать скрипт и запустить его по таймауту
#!/bin/bash
BACKUP_DIR="/home/$USER/backups"
DB_NAME="tms"
DB_USER="hw_losenok"
TIMESTAMP=$(date +"%Y%m%d_%H%M%S")
BACKUP_FILE="${BACKUP_DIR}/${DB_NAME}_${TIMESTAMP}.sql"

mkdir -p "$BACKUP_DIR"
export PGPASSWORD=" "=

pg_dump -h localhost -p 5432 -U "$DB_USER" -d "$DB_NAME" -F p -f "$BACKUP_FILE"

echo "Резервная копия создана: $BACKUP_FILE"

crontab -e
0 2 * * 1 /home/ilya_losenok/backup_tms.sh
