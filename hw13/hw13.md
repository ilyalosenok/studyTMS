#!/bin/bash

# Настройка ВМ (на основе hw4)
# Этапы были разеделены на функции внутри которых использовались регулярные выражения и проверки
# 1. Настройка времени и NTP
setup_time_and_ntp() {
    echo "Настройка даты, времени, часового пояса и NTP."
# Регулярное выражение для проверки формата часового пояса
    timezone="Europe/Minsk"
    if [[ ! "$timezone" =~ ^[A-Za-z]+/[A-Za-z_]+$ ]]; then
        echo "Неверный формат часового пояса"
        return 1
    fi
    sudo timedatectl set-timezone "$timezone"
# Проверка наличия файла timesyncd.conf
    if [ ! -f /etc/systemd/timesyncd.conf ]; then
        echo "Файл /etc/systemd/timesyncd.conf не найден"
        return 1
    fi
    sudo sed -i 's/^#NTP=/NTP=0.pool.ntp.org 1.pool.ntp.org/' /etc/systemd/timesyncd.conf
    sudo sed -i 's/^#FallbackNTP=/FallbackNTP=2.pool.ntp.org 3.pool.ntp.org/' /etc/systemd/timesyncd.conf
    sudo systemctl restart systemd-timesyncd
    check_success "Перезапуск timesyncd"
    echo "Серверы NTP настроены"
}

# 2. Настройка статического IP через netplan
setup_netplan() {
    echo "Настройка статического IP, маршрута и DNS через netplan"
# Поиск файла netplan с помощью регулярного выражения
    netplan_file=$(find /etc/netplan -name "*.yaml" | head -n 1)
    if [ -z "$netplan_file" ]; then
        echo "Файл netplan не найден"
        return 1
    fi
    echo "Найден файл netplan: $netplan_file"
    sudo tee "$netplan_file" <<EOF
network:
  version: 2
  renderer: networkd
  ethernets:
    enp0s3:
      dhcp4: false
      addresses:
        - $ip_address
      routes:
        - to: default
          via: $gateway
      nameservers:
        addresses:
          - 1.1.1.1
          - 8.8.8.8
EOF
    echo "Изменения внесены."
    sudo netplan apply
    echo "Изменения применены."
}

# 3. Создание пользователей и групп 
setup_users_and_groups() {
    echo "Создание пользователей и групп"
# Проверка, существуют ли уже группы
    for group in tms1 tms2; do
        if getent group "$group" > /dev/null; then
            echo "Группа $group уже существует"
        else
            sudo groupadd "$group"
            echo "Создание группы $group"
        fi
    done
# Функция создания пользователя, если его нет
    create_user() {
        local username=$1
        local group=$2
        local password=$3

        if id "$username" &>/dev/null; then
            echo "Пользователь $username уже существует"
        else
            sudo useradd -m -G "$group" -s /bin/bash -c "$comment" "$username"
            echo "Создание пользователя $username"
            echo "$username:$password" | sudo chpasswd
            echo "Установка пароля для $username"
        fi
    }
    create_user "user1" "tms1" "User1" "password"
    create_user "user2" "tms2" "User2" "password"
# Настройка прав через visudo
    if ! sudo grep -q "^%tms1 ALL=(ALL:ALL) /usr/bin/apt update, /usr/bin/apt upgrade" /etc/sudoers; then
        echo "%tms1 ALL=(ALL:ALL) /usr/bin/apt update, /usr/bin/apt upgrade" | sudo EDITOR='tee -a' visudo
        echo "Права для группы tms1"
    else
        echo "Права для tms1 уже настроены"
    fi
    if ! sudo grep -q "^user2 ALL=(ALL) NOPASSWD: /usr/bin/systemctl status" /etc/sudoers; then
        echo "user2 ALL=(ALL) NOPASSWD: /usr/bin/systemctl status *" | sudo EDITOR='tee -a' visudo
        echo "Права для user2"
    else
        echo "Права для user2 уже настроены"
    fi
}

# 4. Создание каталога с наследованием прав
setup_directory() {
    echo "Создание каталога для наследования от группы директории и файла в нем с правами только у владельца"
    local dir_name="forg"
    local file_name="test.me"
# Проверка существует ли директория
    if [ -d "$dir_name" ]; then
        echo "Директория $dir_name уже существует"
    else
        sudo mkdir "$dir_name"
        echo "Создание директории $dir_name"
    fi
    sudo chown -R root:tms1 "$dir_name"
    sudo chmod g+s "$dir_name"
    echo "Установка прав наследования на $dir_name"
# Проверка создан ли файл
    if [ -f "$file_name" ]; then
        echo "Файл $file_name уже существует"
    else
        sudo touch "$file_name"
        echo "Создание файла $file_name"
    fi
    sudo chmod 700 "$file_name"
    echo "Директория и файл с правами только у владельца созданы"
}

# 5. Настройка сетевого доступа по 22 и 80 порту
setup_network() {
    echo "Настройка сетевого доступа по 22 и 80 порту"
# Регулярное выражение для проверки портов
    ports=(22 80)
    for port in "${ports[@]}"; do
        if [[ ! "$port" =~ ^[0-9]+$ ]]; then
            echo "Неверный номер порта: $port"
            return 1
        fi
        sudo ufw allow "$port/tcp"
        echo "Разрешение порта $port"
    done

    sudo ufw default deny incoming
    sudo ufw --force enable
    echo "Настройка UFW"
}
# В конце главный блок с последовательностью выполнения
main() {
    echo "Начало настройки ВМ"

    setup_time_and_ntp
    setup_netplan
    setup_users_and_groups
    setup_directory
    setup_network

    echo "Настройка ВМ успешно завершена"
}

main
